var selectFrame = function () { var frame = frames[Number(range.value)]; cursor.setAttribute('x1', frame.dataset.x); cursor.setAttribute('x2', frame.dataset.x); output.textContent = frame.dataset.summary; apply.setAttribute('href', frame.dataset.url); range.setAttribute('aria-valuetext', frame.dataset.label); }; range.addEventListener('input', selectFrame); range.disabled = false; chart.querySelector('[data-cost-controls]').classList.remove('hidden'); }); } initializeCostCharts(document); // Invalid optional controls must be visible before the browser focuses them. document.addEventListener('invalid', function (event) { var parent = event.target.parentElement; while (parent) { if (parent.tagName === 'DETAILS') parent.open = true; parent = parent.parentElement; } }, true); if (document.querySelector('[data-hosting-calculator]')) { var pendingScenario; async function updateScenario(url, source) { // Only this same-origin, server-rendered calculator supplies replacement markup. if (url.origin !== location.origin || url.pathname !== location.pathname) return; if (pendingScenario) pendingScenario.abort(); var request = new AbortController(); pendingScenario = request; var current = document.querySelector('[data-hosting-calculator]'); current.setAttribute('aria-busy', 'true'); try { var response = await fetch(url, { signal: request.signal }); if (!response.ok) throw new Error('Calculator response: ' + response.status); var html = await response.text(); if (pendingScenario !== request) return; var next = new DOMParser().parseFromString(html, 'text/html').querySelector('[data-hosting-calculator]'); if (!next) throw new Error('Calculator markup is missing'); // Keep deliberate disclosures and the active control in the same viewport position. current.querySelectorAll('details[id]').forEach(function (details) { var replacement = next.querySelector('#' + details.id); if (replacement) replacement.open = details.open; }); var top = source.getBoundingClientRect().top; var sourceID = source.id; current.replaceWith(next); initializeCostCharts(next); var replacement = document.getElementById(sourceID); if (replacement) { replacement.focus({ preventScroll: true }); window.scrollBy({ top: replacement.getBoundingClientRect().top - top, behavior: 'instant' }); } url.hash = ''; history.replaceState(history.state, '', url); next.querySelector('[data-update-status]').textContent = 'Comparison updated.'; } catch (error) { if (error.name !== 'AbortError') { var status = current.querySelector('[data-update-status]'); status.classList.remove('sr-only'); status.setAttribute('role', 'alert'); status.textContent = 'Could not update the comparison. Please try again.'; source.insertAdjacentElement('afterend', status); } } finally { if (pendingScenario === request) { pendingScenario = null; document.querySelector('[data-hosting-calculator]').removeAttribute('aria-busy'); } } } document.addEventListener('change', function (event) { if (event.target.id !== 'model') return; // Display the server-owned static reference without changing the scenario. var option = event.target.selectedOptions[0]; document.querySelector('[data-hardware-minimum]').textContent = option.dataset.hardware; document.querySelector('[data-hardware-multi-host]').hidden = option.dataset.multiHost !== 'true'; }); document.addEventListener('submit', function (event) { var form = event.target; if (form.id !== 'calculator') return; event.preventDefault(); var url = new URL(form.action); url.search = new URLSearchParams(new FormData(form, event.submitter || undefined)); var source = event.submitter && !event.submitter.hidden ? event.submitter : document.activeElement; updateScenario(url, source); }); document.addEventListener('click', function (event) { if (event.defaultPrevented || event.button !== 0 || event.metaKey || event.ctrlKey || event.shiftKey || event.altKey) return; var link = event.target.closest('a[data-scenario-link]'); if (!link) return; event.preventDefault(); updateScenario(new URL(link.href), link); }); }

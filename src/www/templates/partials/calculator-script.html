apply.setAttribute('href', frame.dataset.url); range.setAttribute('aria-valuetext', frame.dataset.label); }; range.addEventListener('input', selectFrame); range.disabled = false; selectFrame(); chart.querySelector('[data-cost-controls]').classList.remove('hidden'); }); } initializeCostCharts(document); // Back/forward form restoration can change the range after initialization without an input event. window.addEventListener('pageshow', function () { document.querySelectorAll('[data-cost-range]').forEach(function (range) { range.dispatchEvent(new Event('input')); }); }); // Invalid optional controls must be visible before the browser focuses them. document.addEventListener('invalid', function (event) { var parent = event.target.parentElement; while (parent) { if (parent.tagName === 'DETAILS') parent.open = true; parent = parent.parentElement; } }, true); if (document.querySelector('[data-hosting-calculator]')) { var pendingScenario = null; var updateStatus = document.querySelector('[data-update-status]'); var announcement; document.querySelector('[data-hosting-calculator]').removeAttribute('aria-busy'); }; // The persistent status is already observed; a later task makes repeated messages audible. var announce = function (message) { clearTimeout(announcement); updateStatus.textContent = ''; announcement = setTimeout(function () { updateStatus.textContent = message; }, 100); }; async function updateScenario(url, source) { // Only this same-origin, server-rendered calculator supplies replacement markup. if (url.origin !== location.origin || url.pathname !== location.pathname) return;

var copyStatus = document.getElementById('copy-link-status'); var copyTimeout; var copyRequest = 0; document.querySelectorAll('.heading-anchor').forEach(function (link) { link.setAttribute('aria-label', link.getAttribute('aria-label').replace('Link to section:', 'Copy link to section:')); link.title = 'Copy link to this section'; link.addEventListener('click', async function (event) { if (event.ctrlKey || event.metaKey || event.shiftKey || event.altKey) return; var request = ++copyRequest; clearTimeout(copyTimeout); copyStatus.textContent = ''; var message; try { await navigator.clipboard.writeText(link.href); message = 'Link copied'; } catch { message = 'Could not copy. Copy the link from your browser’s address bar.'; } if (request !== copyRequest) return; copyStatus.textContent = message; copyTimeout = setTimeout(function () { copyStatus.textContent = ''; }, 5000); }); }); var articleNavigation = document.getElementById('article-sections'); if (articleNavigation) { var articleLinks = Array.from(articleNavigation.querySelectorAll('a')); var articleHeadings = articleLinks.map(function (link) { return document.getElementById(link.hash.slice(1)); }); var articleWideScreen = window.matchMedia('(min-width: 1920px)'); var pendingArticleUpdate = false; var updateArticleSection = function () { pendingArticleUpdate = false; if (!articleWideScreen.matches) return; var current = null; articleHeadings.forEach(function (heading) { var offset = parseFloat(getComputedStyle(heading).scrollMarginTop) + 1; if (heading.getBoundingClientRect().top <= offset) current = heading.id; }); articleLinks.forEach(function (link) { if (link.hash === '#' + current) link.setAttribute('aria-current', 'location'); else link.removeAttribute('aria-current'); }); }; var scheduleArticleUpdate = function () { if (pendingArticleUpdate || !articleWideScreen.matches) return; pendingArticleUpdate = true; requestAnimationFrame(updateArticleSection); }; window.addEventListener('scroll', scheduleArticleUpdate, { passive: true }); window.addEventListener('resize', scheduleArticleUpdate); window.addEventListener('hashchange', scheduleArticleUpdate); new ResizeObserver(scheduleArticleUpdate).observe(document.querySelector('.article-page')); updateArticleSection(); }
